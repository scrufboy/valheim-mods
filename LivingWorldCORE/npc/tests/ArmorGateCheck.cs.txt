using System.Linq; using System.Reflection; using HarmonyLib; using LivingWorld.Core;
class T2 { static int Main() {
  var m = typeof(Character).GetMethod("RPC_Damage", BindingFlags.Instance|BindingFlags.NonPublic|BindingFlags.Public, null, new[]{typeof(long), typeof(HitData)}, null);
  var outp = ArmorParity.RpcDamageTranspiler(PatchProcessor.GetOriginalInstructions(m)).ToList();
  System.Console.WriteLine("armor=" + ArmorParity.ArmorGatesPatched + " stagger=" + ArmorParity.StaggerGatesPatched + " backstab=" + ArmorParity.BackstabGatesPatched);
  for (int k = 0; k < outp.Count; k++) { var mi = outp[k].operand as MethodInfo; if (mi != null && (mi.Name=="CountsArmor"||mi.Name=="AlertedForDamage"))
    System.Console.WriteLine("  " + outp[k-2] + " | " + outp[k-1] + " | >> " + outp[k] + " | " + outp[k+1] + " | " + outp[k+2]); }
  return 0; } }
